# Runtime dependencies for the FortiGate Troubleshooting MCP server.
# Requires Python 3.10+ (the tool signatures use PEP 604 `X | None` unions,
# which pydantic evaluates at runtime).
#
#   python3 -m venv .venv
#   . .venv/bin/activate
#   pip install -r requirements.txt

# MCP server framework. Provides FastMCP and the @mcp.tool decorator.
fastmcp>=3.0

# HTTP client used by fortigate_mcp/client.py for all FortiOS REST calls.
httpx>=0.27

# Imported directly in fortigate_mcp/server.py (Field) for tool parameter
# schemas. Comes in transitively via fastmcp, but pinned explicitly because
# the code imports it by name.
pydantic>=2.0

# Loads .env in fortigate_mcp/config.py. The import is wrapped in try/except,
# so it is technically optional -- drop it if you supply FORTIGATE_* purely as
# real environment variables (systemd, container env, etc).
python-dotenv>=1.0

# Verified working together on:
#   fastmcp 3.4.2, httpx 0.28.1, pydantic 2.13.4, python-dotenv 1.2.2
